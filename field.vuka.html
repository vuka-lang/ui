package ui // Field is a form field: its control, label, help and error. It renders the // control its Type names, an HTML input type ("text" by default); Input, // Password, Textarea, Select and Checkbox are the fields of each kind: // // // type Field struct { // Name is the control's name, matching the form struct's form: tag. Name string // Label is shown above the control; omitted when empty. Label string // Type is the HTML input type ("text" default, "email", "number", "date", // "time", "url", "tel"); the dedicated kinds set it for password, // textarea, select and checkbox. Type string // Value is the control's value; for a checkbox, "true" ticks it. Value string // Options are a select's choices. Options []Option // Error is the field's message; when set the control is marked invalid. // The host supplies it (its own validation), since ui does not import the // runtime, where errors are tracked. Error string // Help is a hint shown under the control. Help string Placeholder string Required bool Disabled bool // Rows is a textarea's height; 3 when 0. Rows int // Min, Max and Step bound a number or date control. Min, Max, Step string Autocomplete string Class string // Attrs are extra attributes on the control. Attrs []Attr kind fieldKind `vuka:"-"` } func (f Field) id() string { return "f-" + f.Name } func (f Field) invalid() bool { return f.Error != "" } func (f Field) classes() string { c := "vk-field" if f.kind == kindCheckbox { c += " vk-field--check" } if f.invalid() { c += " vk-field--invalid" } return withClass(c, f.Class) } // control is the class of a control: base, marked when the field is invalid. func (f Field) control(base string) string { if f.invalid() { return base + " " + base + "--invalid" } return base } // common are the attributes every control but the checkbox has, after its // own class (and type or rows). func (f Field) common() []Attr { a := []Attr{{"id", f.id()}, {"name", f.Name}} if f.Required { a = append(a, Attr{"required", true}) } if f.Disabled { a = append(a, Attr{"disabled", true}) } if f.invalid() { a = append(a, Attr{"aria-invalid", "true"}, Attr{"aria-describedby", f.id() + "-error"}) } if f.Autocomplete != "" { a = append(a, Attr{"autocomplete", f.Autocomplete}) } return append(a, f.Attrs...) } func (f Field) inputType() string { if f.Type == "" { return "text" } return f.Type } func (f Field) rows() int { if f.Rows == 0 { return 3 } return f.Rows } // input is the field's control: a textarea, a select or an input. func (f Field) input() Node { switch f.kind { case kindTextarea: return HTML(``, f.control("vk-textarea"), f.rows(), f.common(), nonEmpty(f.Placeholder), f.Value) case kindSelect: var placeholder Node if f.Placeholder != "" { placeholder = HTML(``, f.Value == "", f.Placeholder) } var options []Node for _, o := range f.Options { options = append(options, HTML(``, o.Value, o.Value == f.Value, o.Label)) } return HTML(``, f.control("vk-select"), f.common(), placeholder, options) } return HTML(``, f.control("vk-input"), f.inputType(), f.Value, f.common(), nonEmpty(f.Placeholder), nonEmpty(f.Min), nonEmpty(f.Max), nonEmpty(f.Step), ) }
